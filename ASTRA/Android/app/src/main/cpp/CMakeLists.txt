cmake_minimum_required(VERSION 3.22.1)
project(astra LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ASTRA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(LIBRAW "${ASTRA_ROOT}/Android/third_party/LibRaw")
file(GLOB_RECURSE RAW_SOURCES "${LIBRAW}/src/*.cpp")
if(NOT RAW_SOURCES)
    message(FATAL_ERROR "LibRaw source must be checked out before building")
endif()
add_library(raw STATIC ${RAW_SOURCES})
target_include_directories(raw PUBLIC "${LIBRAW}")
target_compile_definitions(raw PRIVATE LIBRAW_NOTHREADS LIBRAW_NODLL)
target_compile_options(raw PRIVATE -w)
add_library(astra SHARED NativeEngine.cpp "${ASTRA_ROOT}/Core/AstroCore.cpp")
target_include_directories(astra PRIVATE "${ASTRA_ROOT}/Core")
target_link_options(astra PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(astra raw jnigraphics log)
